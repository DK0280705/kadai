\chapter{非線形方程式（代数方程式）の解法}\label{chap:non-linear-equations}
\section{目標}
次の２つの数値解法による非線形方程式の逐次解法アルゴリズムが理解でき、
プログラミングによる問題解法ができる
\begin{itemize}
  \item ニュートン法による非線形方程式の解法(Newton method for non-linear equations)
  \item 2分法による非線形方程式の解法(Bisection method for non-linear equations)
\end{itemize}
\section{理論}
\subsection{ニュートン法による非線形方程式の解法}
非線形方程式$f(x) = 0$
の解を初期値$x_0$を適当に与えて順次$x_1,x_2,\ldots,x_n$を計算して
近似解を求める方法で、テーラー展開の１次近似式をもとにした方法がニュートン法（図3.2参）である。
いま$f(x)$を点$x_{k-1}$でテーラー展開すると
\begin{equation}
f(x)=f(x_{k-1})+f'(x_{k-1})(x - x_{k-1})+ \frac{f^{(2)}(x_{k-1})}{2!}(x-x_{k-1})^2+\cdots
\end{equation}
となり、２次以降の項を無視すると
\begin{equation}\label{eq:taylor-approx}
f(x)\approx f(x_{k-1})+f'(x_{k-1})(x-x_{k-1})
\end{equation}
$f(x)=0$の解を直接求める代わりに近似的に(\ref{eq:taylor-approx})式=0の解を$x_k$として求めると、
\begin{equation}
x_k = x_{k-1} - \frac{f(x_{k-1})}{f'(x_{k-1})}
\end{equation}
である。これを反復使用するのがニュートン法である。

\begin{algorithm}[H]
\caption{ニュートン法のアルゴリズム}
$k \leftarrow 0$\;
\Repeat{$\left| \frac{x_k - x_{k-1}}{x_k} \right| < \varepsilon$}{
$k \leftarrow k + 1$\;
$x_k \leftarrow x_{k-1} - \frac{f(x_{k-1})}{f'(x_{k-1})}$\;
}
\end{algorithm}

\subsection{2分法による非線形方程式の解法}
非線形方程式$f(x)=0$の解が存在すると思われる区間で$f(x)$の符号を調べながら次第に区間の幅を半分ずつ狭めて、
方程式の解を求める方法である。

\begin{algorithm}[H]
\caption{２分法のアルゴリズム}
$k \leftarrow 0$\;
\Repeat{$\varepsilon > h$}{
$k \leftarrow k + 1$\;
$h \leftarrow \frac{h}{2}$\;
\If{$f(x_k)f(x_k + h) < 0$} {
  $x_{k+1} \leftarrow x_k$\;
}
\Else{
  $x_{k+1} \leftarrow x_k + h$\;
}
}
\end{algorithm}

\section{実装}

ニュートン法は、初期値$x_0$から出発し、(\ref{eq:taylor-approx})式を反復適用して収束判定値$\varepsilon$を満たすまで更新を続ける方法である。以下のコードは、Pythonでの実装である。

\subsection{2分法の実装}
2分法は、根を含む区間$[x_0, x_0+h_0]$から出発し、区間幅$h$を半分に狭めながら根の存在する側を判定していく方法である。

\chapter{定積分の解法}\label{chap:definite-integrals}
\section{目標}
次の４つの数値積分のアルゴリズムを理解し、プログラムを書いて問題解決ができる。
\begin{itemize}
  \item 区分求籍法(Definite Integrals using Rectangles)
  \item 台形公式(Trapezoidal Rule)
  \item シンプソンの公式(Simpson's Rule)
  \item ２重積分(Double Integrals)
\end{itemize}
\section{理論}
\subsection{区分求積法による解法}
$f(x)$を閉区間$[a, b]$で定義された有界関数とし、その定積分：
\begin{equation}\label{eq:def-integral}
S = \int_{a}^{b}f(x)dx
\end{equation}
を考える。積分区間$[a,b]$を$n$等分し、各区間を
\begin{equation}
x_k = a + k \cdot h\text{、ただし、 } h = \frac{b-a}{n}
\end{equation}
とする（$h$は刻み幅）。この区間ごとの面積の総和：
\begin{equation}\label{eq:def-int-parts}
S_n = \sum_{k=0}^{n-1}f(x_k)h
\end{equation}
で近似値を求める方法が区分求積法である。
$n$を十分大きくとると、(\ref{eq:def-int-parts})式$S_n$の近似値が(\ref{eq:def-integral})式の$S$の真値に近づく。
ただし、実際に計算機で行う際には$n$がある程度以上になると丸め誤差が累積するため精度には限界がある。

\subsection{台形公式による解法}
台形公式は、区間$[a,b]$を$n$等分し各小区間ごとの台形の面積の総和で定積分の近似値を求める方法で、
\begin{equation}
S_n = \sum_{k=0}^{n-1}\frac{h}{2}(f(x_k)+f(x_{k+1}))\text{、ただし、 } x_k = a + kh = a + k\frac{b-a}{n}
\end{equation}
となる。

\subsection{シンプソンの公式による解法}
シンプソンの(1/3)公式は、関数の3点を通る2次の補間多項式で近似した式を基にしており、区間$[a,b]$を$n$等分（ただし、$n$は偶数）し、
\begin{equation}\label{eq:simpson}
S_n = \sum_{k=0}^{\frac{n}{2}-1}\frac{h}{3}\bigl(f(x_{2k})+4f(x_{2k+1})+f(x_{2k+2})\bigr)\text{、ただし、 } x_k = a+kh = a+k\frac{b-a}{n}
\end{equation}
となる。定積分は小区間$[x_{2k}, x_{2k+2}]$ごとの面積$S_k'$で以下のように表せる。
\begin{equation}\label{eq:simpson-parts}
\int_{a}^{b}f(x)dx = \sum_{k=0}^{\frac{n}{2}-1}S_k'
\end{equation}
まず、区間$[x_{2k}, x_{2k+2}]$の面積$S_k'$：
\begin{equation}\label{eq:simpson-sk}
S_k' = \int_{x_{2k}}^{x_{2k+2}}f(x)dx
\end{equation}
の近似値を、関数$f(x)$の2次近似補間多項式：
\begin{equation}\label{eq:simpson-g}
f(x)\approx g(x) = a_0 + a_1(x-x_{2k}) + a_2(x-x_{2k})(x-x_{2k+1})
\end{equation}
といわれるこの$g(x)$を用いて近似することを考える。$f(x)$がこの2次近似式$g(x)$と3点$(x_{2k}, x_{2k+1}, x_{2k+2})$で$f(x)=g(x)$となるためには
\begin{equation}
\begin{cases}
f(x_{2k}) = g(x_{2k}) = a_0 \\
f(x_{2k+1}) = g(x_{2k+1}) = a_0 + a_1(x_{2k+1}-x_{2k}) = a_0 + a_1 h \\
f(x_{2k+2}) = g(x_{2k+2}) = a_0 + a_1(x_{2k+2}-x_{2k}) \\
\hphantom{f(x_{2k+2}) = g(x_{2k+2}) =}{}+ a_2(x_{2k+2}-x_{2k})(x_{2k+2}-x_{2k+1}) = a_0 + a_1 2h + a_2 2h^2
\end{cases}
\end{equation}
となり、(\ref{eq:simpson-g})式の近似式の係数$\{a_0, a_1, a_2\}$は
\begin{equation}\label{eq:simpson-coeffs}
\begin{cases}
a_0 = f(x_{2k}) \\
a_1 = \dfrac{1}{h}\bigl(f(x_{2k+1}) - f(x_{2k})\bigr) \\
a_2 = \dfrac{1}{2h^2}\bigl(f(x_{2k}) - 2f(x_{2k+1}) + f(x_{2k+2})\bigr)
\end{cases}
\end{equation}
となる。(\ref{eq:simpson-g})式を(\ref{eq:simpson-sk})式の右辺に代入して積分すると、
\begin{equation}
\begin{aligned}
S_k' &= \int_{x_{2k}}^{x_{2k+2}}f(x)dx \approx \int_{x_{2k}}^{x_{2k+2}}g(x)dx \\
&= \frac{h}{3}\bigl(f(x_{2k})+4f(x_{2k+1})+f(x_{2k+2})\bigr)
\end{aligned}
\end{equation}
となり、(\ref{eq:simpson-parts})式よりシンプソンの公式(\ref{eq:simpson})式が導かれる。
シンプソンの公式による定積分は、2次曲線で近似した関数から求めており、一般に台形公式による定積分より近似精度がよい。

\subsection{２重積分}
長方形領域$A\{a\le x\le b,\ c\le y\le d\}$の上で定義された2変数の連続関数$f(x,y)$の2重積分：
\begin{equation}\label{eq:double-integral}
V = \int\!\!\int_{A} f(x,y)\,dx\,dy
\end{equation}
について考える。まず、$x$についての積分を1次元の積分で用いた区分求積法(\ref{eq:def-int-parts})式を適用する。$x$の積分区間$[a,b]$を$n$等分すると、
\begin{equation}
V \approx \int_{c}^{d}\left(\sum_{i=0}^{n-1}f(x_i,y)h_x\right)dy\text{、ただし、 } h_x = \frac{b-a}{n}
\end{equation}
となり、さらに$y$の積分区間$[c,d]$を$m$等分して$y$について区分求積法を適用すると、
\begin{equation}\label{eq:double-integral-approx}
V = \int\!\!\int_{A} f(x,y)\,dx\,dy \approx \sum_{i=0}^{n-1}\sum_{j=0}^{m-1}f(x_i,y_j)h_x h_y
\end{equation}
ただし、
\begin{equation}
h_x = \frac{b-a}{n}\text{、}\quad h_y = \frac{d-c}{m}
\end{equation}
となる。この方法は、1次元の積分方法である区分求積法を2次元に拡張した方法である。
この区分求積法を用いた2重積分は、各小区間$\{x_i\le x\le x_i+h_x,\ y_j\le y\le y_j+h_y\}$ごとの体積を底面積$h_xh_y$、高さ$f(x_i,y_j)$の立方体$V_{ij}$：
\begin{equation}
V_{ij} = f(x_i,y_j)h_x h_y
\end{equation}
で近似し、全ての領域においてこの立方体の体積の総和で近似することで、2重積分の近似式(\ref{eq:double-integral-approx})式：
\begin{equation}
V \approx \sum_{i=0}^{n-1}\sum_{j=0}^{m-1}V_{ij} = \sum_{i=0}^{n-1}\sum_{j=0}^{m-1}f(x_i,y_j)h_x h_y
\end{equation}
が得られる。

\subsection{GPU（コンピュートシェーダー）による2重積分の高速化}\label{sec:gpu-double-integral}
(\ref{eq:double-integral-approx})式の2重和は、格子点$(x_i, y_j)$ごとの評価が互いに独立であるため、GPU上で大規模に並列化できる。そこで、最適化の一例として、応用課題2.6の被積分関数$f(x,y)=x^2y^2$に特化したGLSLコンピュートシェーダーを\texttt{moderngl}経由で実行し、CPU（NumPy）実装との速度比較を行った。

シェーダー側では、各スレッドが1つの格子点$(i,j)$の値$f(x_i,y_j)h_xh_y$を計算し、ワークグループ内で共有メモリを用いた並列リダクション（総和）を行う。ワークグループごとの部分和のみをCPU側に転送し、最後にその小さな配列（ワークグループ数個）をNumPyで合計することで、$n\times m$個の値をCPUへ転送するコストを避けている。

\bigskip\noindent\textbf{GPU（コンピュートシェーダー）による高速化}\medskip
